<script>
  // FAKE demo version so it works offline
  let currentUser = null;
  let profiles = [
    {name:"Alice", bio:"I love hiking!", photoURL:"https://placekitten.com/300/300"},
    {name:"Bob", bio:"Coffee addict ☕", photoURL:"https://placekitten.com/301/301"},
    {name:"Carol", bio:"Music is life 🎵", photoURL:"https://placekitten.com/302/302"}
  ];
  let index = 0;

  function signup() {
    currentUser = emailInput.value || "user@example.com";
    showProfile();
  }
  function login() {
    currentUser = emailInput.value || "user@example.com";
    showSwipe();
  }

  function saveProfile() {
    alert("Profile saved! Now you can swipe.");
    showSwipe();
  }

  function showCard() {
    if(!profiles.length) return;
    cardImg.src = profiles[index].photoURL;
    cardName.innerText = profiles[index].name;
    cardBio.innerText = profiles[index].bio;
  }

  function like() { 
    alert("You liked " + profiles[index].name);
    next();
  }
  function pass() { 
    next(); 
  }
  function next() { 
    index = (index + 1) % profiles.length; 
    showCard(); 
  }

  function hideAll() {
    authBox.style.display = profileBox.style.display = swipeBox.style.display = "none";
  }
  function showProfile() {
    hideAll();
    profileBox.style.display = "block";
  }
  function showSwipe() {
    hideAll();
    swipeBox.style.display = "block";
    showCard();
  }
</script>
